SET SQL DIALECT 1;

CREATE DOMAIN DOM_DESCRICAO AS VARCHAR(80) NOT NULL;

CREATE TABLE TB_BAIRRO 
(
  CODBAIRRO	INTEGER NOT NULL,
  DSCBAIRRO	DOM_DESCRICAO,
CONSTRAINT TB_BAIRRO_PK PRIMARY KEY (CODBAIRRO),
CONSTRAINT UK_BAIRRO UNIQUE (DSCBAIRRO)
);

CREATE TABLE TB_ENDERECO 
(
  CODENDERECO	INTEGER NOT NULL,
  DSCENDERECO	DOM_DESCRICAO,
CONSTRAINT TB_ENDERECO_PK PRIMARY KEY (CODENDERECO),
CONSTRAINT UK_ENDERECO UNIQUE (DSCENDERECO)
);

CREATE TABLE TB_ESTADO 
(
  CODESTADO	INTEGER NOT NULL,
  DSCESTADO	DOM_DESCRICAO,
CONSTRAINT TB_ESTADO_PK PRIMARY KEY (CODESTADO),
CONSTRAINT UK_ESTADO UNIQUE (DSCESTADO)
);

CREATE TABLE TB_CIDADE 
(
  CODCIDADE	INTEGER NOT NULL,
  DSCCIDADE	DOM_DESCRICAO,
  CODESTADO	INTEGER NOT NULL,
CONSTRAINT TB_CIDADE_PK PRIMARY KEY (CODCIDADE),
CONSTRAINT UK_CIDADE UNIQUE (DSCCIDADE, CODESTADO)
);

CREATE TABLE TB_CLIENTE 
(
  CODCLIENTE	INTEGER NOT NULL,
  NOME	DOM_DESCRICAO,
  CPF	VARCHAR(12) NOT NULL,
  RG	VARCHAR(12),
  NUMERO	INTEGER NOT NULL,
  DATA_NASCIMENTO	DATE,
  DATA_CADASTRO	DATE NOT NULL.
  CODENDERECO	INTEGER NOT NULL,
  CODBAIRRO	INTEGER NOT NULL,
  CODCIDADE	INTEGER NOT NULL,
CONSTRAINT TB_CLIENTE_PK PRIMARY KEY (CODCLIENTE),
CONSTRAINT UK_CLIENTE UNIQUE (NOME, CPF)
);

CREATE TABLE TB_COR 
(
  CODCOR	INTEGER NOT NULL,
  DSCCOR	DOM_DESCRICAO,
CONSTRAINT TB_COR_PK PRIMARY KEY (CODCOR),
CONSTRAINT UK_COR UNIQUE (DSCCOR)
);

CREATE TABLE TB_MARCA 
(
  CODMARCA	INTEGER NOT NULL,
  DSCMARCA	DOM_DESCRICAO,
CONSTRAINT TB_MARCA_PK PRIMARY KEY (CODMARCA),
CONSTRAINT UK_MARCA UNIQUE (DSCMARCA)
);

CREATE TABLE TB_VEICULO 
(
  CODVEICULO	INTEGER NOT NULL,
  DSCVEICULO	DOM_DESCRICAO,
  CODMARCA	INTEGER NOT NULL,
  ANOFAB	INTEGER NOT NULL,
  ANOMODELO	INTEGER NOT NULL,
  CHASSI	INTEGER NOT NULL,
  PLACA	VARCHAR(8),
  CODCOR	INTEGER NOT NULL,
  VALORCOMPRA	FLOAT DEFAULT 0,
  VALORVENDA	FLOAT DEFAULT 0,
CONSTRAINT TB_VEICULO_PK PRIMARY KEY (CODVEICULO),
CONSTRAINT UK_VEICULO UNIQUE (DSCVEICULO, CODMARCA, CHASSI)
);

CREATE TABLE TB_VENDEDOR 
(
  CODVENDEDOR	INTEGER NOT NULL,
  NOME	DOM_DESCRICAO,
  COMISSAO	FLOAT,
  DATA_ADMISSAO Date,
CONSTRAINT TB_VENDEDOR_PK PRIMARY KEY (CODVENDEDOR),
CONSTRAINT UK_VENDEDOR UNIQUE (NOME)
);

CREATE TABLE TB_COMPRA 
(
  CODCOMPRA	INTEGER NOT NULL,
  DTCOMPRA	DATE NOT NULL,
  CODVEICULO	INTEGER NOT NULL,
  CODCLIENTE	INTEGER NOT NULL,
CONSTRAINT TB_COMPRA_PK PRIMARY KEY (CODCOMPRA)
);

CREATE TABLE TB_VENDA 
(
  CODVENDA	INTEGER NOT NULL,
  DTVENDA	DATE NOT NULL,
  CODVENDEDOR	INTEGER NOT NULL,
  CODVEICULO	INTEGER NOT NULL,
  CODCLIENTE	INTEGER NOT NULL,
CONSTRAINT TB_VENDA_PK PRIMARY KEY (CODVENDA)
);

ALTER TABLE TB_COMPRA ADD FOREIGN KEY (CODCLIENTE) REFERENCES TB_CLIENTE (CODCLIENTE);
